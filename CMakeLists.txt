cmake_minimum_required(VERSION 3.20)
project(bolt_multiplier VERSION 0.3.1 LANGUAGES CXX)

include(CTest)
option(BOLT_RULES_ONLY "Build standalone core tests without the OpenRAC SDK" OFF)

if(NOT BOLT_RULES_ONLY)
  # The SDK can be supplied explicitly, or fetched from sdk.lock by tools/fetch_sdk.py.
  if(NOT OpenRACMod_DIR)
    if(DEFINED OPENRAC_SDK)
      set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
    else()
      file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" lock)
      string(JSON sdk_version GET "${lock}" version)
      set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}/cmake")
      if(NOT EXISTS "${OpenRACMod_DIR}/OpenRACModConfig.cmake")
        message(FATAL_ERROR "OpenRAC SDK not found. Run: python tools/fetch_sdk.py")
      endif()
    endif()
  endif()
  find_package(OpenRACMod 1.12 REQUIRED)
  openrac_add_mod(bolt_multiplier MANIFEST openrac-mod.json SOURCES src/mod.cpp src/rules.cpp)
  target_include_directories(bolt_multiplier PRIVATE src)
  set_target_properties(bolt_multiplier PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  if(NOT MSVC)
    target_compile_options(bolt_multiplier PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

if(BUILD_TESTING)
  add_executable(bolt_multiplier_rules_test tests/rules_test.cpp src/rules.cpp)
  target_include_directories(bolt_multiplier_rules_test PRIVATE src)
  set_target_properties(bolt_multiplier_rules_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  if(NOT MSVC)
    target_compile_options(bolt_multiplier_rules_test PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME bolt_multiplier_rules COMMAND bolt_multiplier_rules_test)
endif()
